#Generators
SquareGenerator := function(n,imin,imax)
  local list,i,ints;
  list := [[]];
  ints := [imin..imax];
  for i in [1..n] do
    if i in ints
	  then Add(list[1],i);
	    Add(list[1],-i);
	else Add(list,[i,-i]);
	fi;
  od;
  return Bipartition(list);
end;

DotGenerator := function(n,ints)
  local list,i;
  list := [];
  for i in [1..n] do
    if i in ints
	  then Add(list,[i]);
	    Add(list,[-i]);
	else Add(list,[i,-i]);
	fi;
  od;
  return Bipartition(list);
end;

SymGenerator := function(n,int)
  local list,i;
  list := [[int,-(int+1)],[-int,int+1]];
  for i in [1..int-1] do
    Add(list,[i,-i]);
  od;
  for i in [int+2..n] do
    Add(list,[i,-i]);
  od;
  return Bipartition(list);
end;

CupGenerator := function(n,int)
  local list,i;
  list := [[int,int+1],[-int,-(int+1)]];
  for i in [1..int-1] do
    Add(list,[i,-i]);
  od;
  for i in [int+2..n] do 
    Add(list,[i,-i]);
  od;
  return Bipartition(list);
end;

pRoLeft := function(n,i)
  return SymGenerator(n,i)*DotGenerator(n,[i]);
end;

pRoRight := function(n,i)
  return DotGenerator(n,[i])*SymGenerator(n,i);
end;

DiagramGenerators := function(n)
  local syms,squares,prookleft,prookright,cups,dots,i;
  syms := []*(n-1);
  squares := []*(n-1);
  prookleft := []*(n-1);
  prookright := []*(n-1);
  cups := []*(n-1);
  dots := []*n;
  for i in [1..(n-1)] do
    squares[i] := SquareGenerator(n,i,i+1);
    syms[i] := SymGenerator(n,i);
    cups[i] := CupGenerator(n,i);
    dots[i] := DotGenerator(n,[i]);
    prookleft[i] := pRoLeft(n,i);
    prookright[i] := pRoRight(n,i);
  od;
  dots[n] := DotGenerator(n,[n]);
  return [cups,dots,syms,squares,prookleft,prookright];
end;

#Temperley-Lieb
TLFact := function(bipartition) 
  local factors,k,blocks,n,pair,midfactors,botfactors,oddfact,evenfact,topfact,botfact,i;
  blocks := List(ExtRepOfObj(bipartition));
  n := DegreeOfBipartition(bipartition);
  factors := [];
  midfactors := [];
  topfact := [];
  botfact := [];
  botfactors := [];
  for pair in blocks do
    if pair[1] > 0 and pair[2] > 0 then 
	  oddfact := [];
	  evenfact := [];
	  for i in [pair[1]..pair[2]-1] do
		if (i mod 2) = (pair[1] mod 2) then
		  Add(evenfact,CupGenerator(n,i));
		elif (i mod 2) <> (pair[1] mod 2) then
		  Add(oddfact,CupGenerator(n,i));
		fi;
	  od;
	  Append(factors,oddfact);
	  Append(factors,evenfact);
	elif pair[1] > 0 and pair[2] < 0 then
	  if pair[1] <> -pair[2] then
		oddfact := [];
	    evenfact := [];
		if pair[1] > -pair[2] then
	      for i in [-pair[2]..pair[1]-1] do
		    if (i mod 2) = (pair[1] mod 2) then
		      Add(evenfact,CupGenerator(n,i));
		    elif (i mod 2) <> (pair[1] mod 2) then
		      Add(oddfact,CupGenerator(n,i));
		    fi;
	      od;
		  Append(botfact,evenfact);
		  Append(botfact,oddfact);
		elif pair[1] < -pair[2] then
		  for i in [pair[1]..-pair[2]-1] do
		    if (i mod 2) = (pair[1] mod 2) then
		      Add(evenfact,CupGenerator(n,i));
		    elif (i mod 2) <> (pair[1] mod 2) then
		      Add(oddfact,CupGenerator(n,i));
		    fi;
	      od;
		  Append(topfact,evenfact);
		  Append(topfact,oddfact);
		fi;
	  fi;
	else
	  oddfact := [];
	  evenfact := [];
	  for i in [-pair[1]..-pair[2]-1] do
		if (i mod 2) = (-pair[1] mod 2) then
		  Add(evenfact,CupGenerator(n,i));
		elif (i mod 2) <> (-pair[1] mod 2) then
		  Add(oddfact,CupGenerator(n,i));
		fi;
	  od;
	  Append(botfactors,evenfact);
	  Append(botfactors,oddfact);
	fi;
  od;
  factors := Reversed(factors);
  Append(factors,Reversed(topfact));
  Append(factors,botfact);
  Append(factors,botfactors);
  return factors;
end;

#use Dyck path sampling; https://arxiv.org/pdf/1603.06290
TLRandom := function(n)
  local a,b,h,num,den,dyck,step,stack,blocks,i,p,N,BoundaryVertex;
  N := 2*n;
  
  #generate a uniformly random Dyck path of semilength k
  #a = number of up-steps remaining
  #b = number of down-steps remaining
  #h = b - a = current height
  #conditional on the current prefix, the probability that the next step is an up-step is
  #P(U) = a(h+2)/((h+1)(a+b)); uses Bertrand's Ballot Theorem
  dyck := [];
  a := n;
  b := n;
  h := 0;
  while a + b > 0 do
    if a = 0 then
	  Add(dyck, -1);
	  b := b - 1;
	  h := h - 1;
	elif h = 0 then
	  Add(dyck, 1);
	  a := a - 1;
	  h := h + 1;
	else
	  num := a * (h + 2);
	  den := (h + 1) * (a + b);
	  if Random(1, den) <= num then
	    Add(dyck, 1);
		a := a - 1;
		h := h + 1;
	  else
	    Add(dyck, -1);
		b := b - 1;
		h := h - 1;
	  fi;
    fi;
  od;

  #convert dyck path to TL diagram
  BoundaryVertex := function(pos)
    if pos <= n then
	  return pos;
	else
	  return -(N-pos+1);
    fi;
  end;
  
  stack := [];
  blocks := [];
  for i in [1 .. N] do
    step := dyck[i];
	if step = 1 then
	  Add(stack, i);
	else
	  p := Remove(stack);
	  Add(blocks,[BoundaryVertex(p),BoundaryVertex(i)]);
    fi;
  od;

  return Bipartition(blocks);
end;

pRoFact := function(bipartition)
  local factors,k,blocks,n,pair,botfactors,oddfact,evenfact,topfact,botfact,i;
  blocks := List(ExtRepOfObj(bipartition));
  n := DegreeOfBipartition(bipartition);
  factors := [];
  topfact := [];
  botfact := [];
  botfactors := [];
  for pair in blocks do
    if Size(pair) = 1 then 
	  if pair[1] > 0 then
	    if pair[1] = n then
		  Add(factors,pRoLeft(n,n-1));
		  Add(factors,pRoRight(n,n-1));
		else
		  Add(factors,pRoRight(n,pair[1]));
		  Add(factors,pRoLeft(n,pair[1]));
		fi;
	  else 
	    if pair[1] = -n then
		  Add(botfactors,pRoLeft(n,n-1));
		  Add(botfactors,pRoRight(n,n-1));
		else 
		  Add(botfactors,pRoRight(n,-pair[1]));
		  Add(botfactors,pRoLeft(n,-pair[1]));
	    fi;
	  fi;
	else
	  if pair[1] <> -pair[2] then
		oddfact := [];
		if pair[1] > -pair[2] then
	      for i in [-pair[2]..pair[1]-1] do
		    Add(oddfact,pRoRight(n,i));
	      od;
		  Append(botfact,Reversed(oddfact));
		elif pair[1] < -pair[2] then
		  for i in [pair[1]..-pair[2]-1] do
		    Add(oddfact,pRoLeft(n,i));
	      od;
		  Append(topfact,Reversed(oddfact));
		fi;
	  fi;
	fi;
  od;
  Append(factors,Reversed(topfact));
  Append(factors,botfact);
  Append(factors,botfactors);
  return factors;
end;

#randomly choose top/bottom half diagram with fixed number of through strands k
pRoRandomk := function(n,k)
  local r,subset,next,remaining,count;
  if k = 0 then
    return [];
  fi;

  r := Random(1, Binomial(n,k));
  subset := [];
  next := 1;
  remaining := k;

  while remaining > 0 do
    #number of subsets whose next selected entry is 'next'.
    count := Binomial(n - next, remaining - 1);
    if r <= count then
      Add(subset, next);
      remaining := remaining - 1;
    else
      r := r - count;
    fi;
	next := next + 1;
  od;
  return subset;
end;

pRoRandom := function(n)
  local total,r,k,numberOfRankK,top,bottom,usedTop,usedBottom,blocks,i;

  # |pRo(n)| = Sum_{k=0}^n Binomial(n,k)^2 = Binomial(2*n,n).
  total := Binomial(2 * n, n);
  
  #randomly choose nubmer of through strands with probability Binomial(n,k)^2 / Binomial(2*n,n).
  r := Random(1, total);
  k := 0;
  numberOfRankK := Binomial(n, k)^2;

  while r > numberOfRankK do
      r := r - numberOfRankK;
      k := k + 1;
      numberOfRankK := Binomial(n, k)^2;
  od;

  #conditional on rank k, choose the top and bottom k-subsets
  #independently and uniformly.
  top := pRoRandomk(n, k);
  bottom := pRoRandomk(n, k);

  usedTop := BlistList([1..n], top);
  usedBottom := BlistList([1..n], bottom);

  blocks := [];

  #join selected vertices in increasing order
  for i in [1..k] do
    Add(blocks, [top[i], -bottom[i]]);
  od;

    #every unused top or bottom vertex is an isolated singleton
  for i in [1..n] do
    if not usedTop[i] then
      Add(blocks, [i]);
    fi;
      if not usedBottom[i] then
        Add(blocks, [-i]);
      fi;
  od;
  return Bipartition(blocks);
end;

#use Motzkin path sampling; https://arxiv.org/abs/1603.06290, https://arxiv.org/abs/2606.13151
MoRandom := function(n)
  local N,weights,w,total,r,k,i,a,b,h,den,num,dyck,step,remainingDyck,remainingPositions,path,j,stack,blocks,p,q,BoundaryVertex;
  N := 2 * n;

  #Motzkin number weights w_k = Binomial(N, 2k)*Catalan(k)
  #w_{k+1} = w_k(N-2k)(N-2k-1)/((k+1)(k+2))
  weights := [];
  w := 1;
  Add(weights, w); #w_0
  for i in [0 .. n - 1] do
    w := QuoInt(w*(N-2*i)*(N-2*i-1),(i+1)*(i+2));
	Add(weights, w); #weights[i+2] = w_{i+1}
  od;
  total := Sum(weights);
  r := Random(1, total);
  k := 0;
  while r > weights[k+1] do
    r := r - weights[k+1];
	k := k + 1;
  od;

  #generate a uniformly random Dyck path of semilength k
  #a = number of up-steps remaining
  #b = number of down-steps remaining
  #h = b - a = current height
  #conditional on the current prefix, the probability that the next step is an up-step is
  #P(U) = a(h+2)/((h+1)(a+b)); uses Bertrand's Ballot Theorem
  dyck := [];
  a := k;
  b := k;
  h := 0;
  while a + b > 0 do
    if a = 0 then
	  Add(dyck, -1);
	  b := b - 1;
	  h := h - 1;
	elif h = 0 then
	  Add(dyck, 1);
	  a := a - 1;
	  h := h + 1;
	elif b = 0 then
	  Add(dyck, 1);
	  a := a - 1;
	  h := h + 1;
	else
	  num := a*(h+2);
	  den := (h+1)*(a+b);
	  if Random(1, den) <= num then
	    Add(dyck, 1);
		a := a - 1;
		h := h + 1;
	  else
	    Add(dyck, -1);
		b := b - 1;
		h := h - 1;
	  fi;
	fi;
  od;

  #insert random horizontal steps to make a Motzkin path
  path := [];
  j := 1;
  remainingDyck := 2*k;
  remainingPositions := N;
  while remainingPositions > 0 do
    if remainingDyck > 0 and Random(1,remainingPositions) <= remainingDyck then
	  Add(path, dyck[j]);
	  j := j + 1;
	  remainingDyck := remainingDyck - 1;
	else
	  Add(path,0);
    fi;
	remainingPositions := remainingPositions - 1;
  od;

  #convert Motzkin path to Motzkin diagram
  BoundaryVertex := function(pos)
    if pos <= n then
	  return pos;
    else
	  return -(N-pos+1);
	fi;
  end;
  stack := [];
  blocks := [];
  for i in [1 .. N] do
    step := path[i];
	if step = 1 then
	  Add(stack, i);
	elif step = 0 then
	  Add(blocks, [BoundaryVertex(i)]);
	else
	  p := Remove(stack);
	  q := i;
	  Add(blocks, [BoundaryVertex(p),BoundaryVertex(q)]);
	fi;
  od;
  
  return Bipartition(blocks);
end;

#convert TL diagram to pPa diagram
TLtopPa := function(bipartition)
  local blocks,degree,n,nrverts,partner,visited,ToIndex,TopPaVertex,FixedPartner,block,a,b,i,v,w,pblock,result;
  blocks := ExtRepOfObj(bipartition);
  degree := DegreeOfBipartition(bipartition);
  n := degree / 2;
  nrverts := 4 * n;
  
  #convert to one line
  #i  -> i
  #-i  -> 2n+i
  ToIndex := function(x)
    if x > 0 then
      return x;
    else
      return 2 * n - x;
    fi;
  end;

  #contract TL vertex to the corresponding pPa vertex:
  #2i-1, 2i -> i
  #-(2i-1), -2i    -> -i
  TopPaVertex := function(index)
    if index <= 2 * n then
      return QuoInt(index + 1, 2);
    else
      return -QuoInt(index - 2 * n + 1, 2);
    fi;
  end;

  #pair TL vertices: 1 with 2, 3 with 4, ...
  FixedPartner := function(index)
    if index mod 2 = 1 then
      return index + 1;
    else
      return index - 1;
    fi;
  end;

  # partner[v] is the other endpoint of the TL string containing vertex v
  partner := ListWithIdenticalEntries(nrverts, 0);
  
  #sort
  for block in blocks do
    a := ToIndex(block[1]);
    b := ToIndex(block[2]);
    partner[a] := b;
    partner[b] := a;
  od;

  visited := ListWithIdenticalEntries(nrverts, false);
  result := [];

  for i in [1 .. nrverts] do
    if not visited[i] then
      pblock := [];
      v := i;
	  
	  #run over all vertices that end up in the same connected component in the pPa diagram
      repeat
        # v and w are the two TL points that contract to one vertex of the planar partition diagram
        w := FixedPartner(v);
        visited[v] := true;
        visited[w] := true;
        Add(pblock, TopPaVertex(v));
        #follow the TL edge from the fixed partner
        v := partner[w];
      until visited[v];

      Add(result, pblock);
    fi;
  od;
  return Bipartition(result);
end;

#setting up Fenwick Tree
FenwickTree := function(n)
  return rec(n := n, bit := List([1..n], i -> 0));
end;

FenwickAdd := function(F, idx, delta)
  while idx <= F.n do
    F.bit[idx] := F.bit[idx] + delta;
    idx := idx + 2^(Valuation(idx,2));
  od;
end;

FenwickSum := function(F, idx)
  local s;
  s := 0;
  while idx > 0 do
    s := s + F.bit[idx];
    idx := idx - 2^(Valuation(idx,2));
  od;
  return s;
end;

#factor permutation in its one-line form
PermFact := function(g)
  local n,p,posOfValue,F,x,pos,rank,word,j;
  n := LargestMovedPoint(g);
  if n = 0 then
    return [];
  fi;
  p := List([1..n], i -> i^g);

  # position of each value in one-line notation
  posOfValue := List([1..n], i -> 0);
  for pos in [1..n] do
    posOfValue[p[pos]] := pos;
  od;

  #Fenwick tree initially contains all positions
  F := FenwickTree(n);
  for pos in [1..n] do
    FenwickAdd(F, pos, 1);
  od;

  word := [];
  for x in Reversed([1..n]) do
    pos := posOfValue[x];
    rank := FenwickSum(F, pos);
    for j in [rank .. x-1] do
	  Add(word, j);
    od;
    FenwickAdd(F, pos, -1);
  od;
  return Reversed(word);
end;

#convert bipartition form to one-line permutation form
BipartitionToPerm := function(bipartition,n)
  local p,block;
  p := List([1..n], i -> 0);
  for block in bipartition do
    p[block[1]] := -block[2];
  od;
  return PermList(p);
end;

SymFact := function(bipartition)
  local i,word,factors,syms,n;
  n := DegreeOfBipartition(bipartition);
  syms := []*(n-1);
  for i in [1..(n-1)] do
	syms[i] := SymGenerator(n,i);
  od;
  word := PermFact(BipartitionToPerm(ExtRepOfObj(bipartition),n));
  factors := [];
  for i in word do
    Add(factors,syms[i]);
  od;
  return Reversed(factors);
end;

SymRandom := function(n)
  local images,i,j,temp,blocks;
  #Fisher-Yates shuffle of [1..n]
  images := [1..n];
  for i in Reversed([2..n]) do
    j := Random(1, i);
    temp := images[i];
    images[i] := images[j];
    images[j] := temp;
  od;
  #the permutation i -> images[i] corresponds to the blocks [i, -images[i]].
  blocks := List([1..n], i -> [i, -images[i]]);
  return blocks;
end;

    term := term * (n - k)^2 / (k + 1);
    Add(weights, term);
    total := total + term;
  od;
  r := Random(1, total);
  cumulative := 0;
  for k in [0 .. n] do
    cumulative := cumulative + weights[k + 1];
    if r <= cumulative then
      return k;
    fi;
  od;
end;

RoBrRandom := function(n)
  local N,telephone,points,blocks,m,i,r,j,x,y;
  N := 2 * n;

  #telephone[m + 1] is the number I_m of partitions of m labelled points into singleton and pair blocks
  #I_0 = I_1 = 1,
  #I_m = I_(m-1) + (m-1) I_(m-2).
  telephone := [1, 1];
  for m in [2 .. N] do
    Add(telephone,telephone[m]+(m-1)*telephone[m-1]);
  od;
  
  points := Concatenation([1 .. n],List([1 .. n], i -> -i));
  blocks := [];
  m := N;
  while m > 0 do
    x := points[m];
    if m = 1 then
      Add(blocks, [x]);
      m := 0;
    else
      r := Random(1, telephone[m + 1]);
	  if r <= telephone[m] then
	    Add(blocks, [x]);
		m := m - 1;
	  else
	    j := Random(1, m - 1);
		y := points[j];
		Add(blocks, [x, y]);
		points[j] := points[m - 1];
		m := m - 2;
      fi;
    fi;
  od;
  return Bipartition(blocks);
end;

PaRandom := function(n)
  local N,bell,r,s,total,points,SplitRandomSubset,SamplePartition;
  N := 2 * n;
  #compute Bell numbers B_r = bell[r+1]
  bell := [1];
  for r in [1 .. N] do
    total := 0;
	for s in [0 .. r - 1] do
	  total := total + Binomial(r-1,s)*bell[s+1];
    od;
	Add(bell, total);
  od;

  #choose a random q element subset of a list L
  SplitRandomSubset := function(L, q)
    local A,len,i,j,temp,chosen,unchosen;
	A := ShallowCopy(L);
	len := Length(A);
	
	#partial Fisher-Yates shuffle
	for i in [1 .. q] do
	  j := Random(i, len);
	  temp := A[i];
	  A[i] := A[j];
	  A[j] := temp;
    od;
	if q = 0 then
	  chosen := [];
	else
	  chosen := A{[1 .. q]};
	fi;
	if q = len then
	  unchosen := [];
	else
	  unchosen := A{[q+1 .. len]};
	fi;
	return [chosen, unchosen];
  end;
	
  #recursively sample
  SamplePartition := function(L)
    local len,first,tail,k,weight,cumulative,ticket,chosenK,split,block,remainder,result;
	  len := Size(L);
	  if len = 0 then
	    return [];
	  fi;
	  first := L[1];
	  if len = 1 then
	    return [[first]];
	  fi;
	  tail := L{[2 .. len]};
	  ticket := Random(1, bell[len + 1]);
	  cumulative := 0;
	  chosenK := fail;
	  
	  #one block of size k has Binomial(len-1,k-1) elements apart from first
	  #partition the rest
      for k in [1 .. len] do
	    weight := Binomial(len-1,k-1)*bell[len-k+1];
		cumulative := cumulative + weight;
		if ticket <= cumulative then
		  chosenK := k;
		break;
        fi;
      od;
	  split := SplitRandomSubset(tail, chosenK - 1);
	  block := Concatenation([first], split[1]);
	  remainder := split[2];
	  result := SamplePartition(remainder);
	  Add(result, block, 1);
	  return result;
  end;

  points := Concatenation([1 .. n],List([1 .. n], i -> -i));
  return Bipartition(SamplePartition(points));
end;

PaFactCount := function(bipartition)
  local n,dots,syms,squares,i,j,count,blocks,block,topreq,botreq,listtop,word,listbot,factors,counter,topfactors,botfactors,topprod,botprod,midfactors,botsym,flatbotprod,flatbotreq,flattopprod,flattopreq,topsym,topbotprod,bottopprod,fact,factbotsym,facttopsym,s,tmp;
  blocks := List(ExtRepOfObj(bipartition));
  n := DegreeOfBipartition(bipartition);

  topreq := [];
  botreq := [];
  for block in blocks do
    listtop := [];
	listbot := [];
	for i in block do
	  if i > 0
	    then Add(listtop,i);
	  else Add(listbot,i);
	  fi;
	od;
	Add(topreq,listtop);
	Add(botreq,listbot);
  od;
  
  count := 0;
  counter := 1;
  topfactors := [];
  topprod := [];
  bottopprod := [];
  for block in topreq do
    Add(bottopprod,[(counter)..(counter-1+Size(block))]);
    if not IsEmpty(block) then
	  if Size(block) = 1 then
		Add(topprod,[counter]);
		counter := counter + Size(block);
      else
		for i in [(counter)..(counter-2+Size(block))] do
		  count := count + 1;
		od;
	    Add(topprod,[(counter)..(counter-1+Size(block))]);
	    counter := counter + Size(block);
	  fi;
	fi;
  od;
  counter := 1;
  botfactors := [];
  botprod := [];
  topbotprod := [];
  for block in botreq do
    Add(topbotprod,[(counter)..(counter-1+Size(block))]);
    if not IsEmpty(block) then
	  if Size(block) = 1 then
		Add(botprod,[-counter]);
		counter := counter + Size(block);
      else 
	    for i in [(counter)..(counter-2+Size(block))] do
		  count := count + 1;
		od;
	    Add(botprod,-[(counter)..(counter-1+Size(block))]);
	    counter := counter + Size(block);
	  fi;
	fi;
  od;

  flattopprod := Flat(topprod);
  flattopreq := Flat(topreq);
  topsym := [];
  flatbotprod := Flat(botprod);
  flatbotreq := Flat(botreq);
  botsym := [];
  for i in [1..n] do
	Add(topsym,[-flattopprod[i],flattopreq[i]]);
	Add(botsym,[-flatbotprod[i],flatbotreq[i]]); 
  od;
  count := count + Size(PermFact(BipartitionToPerm(ExtRepOfObj(Bipartition(topsym)),n)));
  count := count + Size(PermFact(BipartitionToPerm(ExtRepOfObj(Bipartition(botsym)),n)));
  
  midfactors := [];
  for i in [1..Size(blocks)] do
    if IsEmpty(topbotprod[i]) and IsEmpty(bottopprod[i]) 
	  then continue;
	elif IsEmpty(topbotprod[i]) then
	  fact := [];
	  for j in [1..Size(bottopprod[i])] do
	    Add(fact,[bottopprod[i][j]]);
	  od;
	  Append(midfactors,fact);
	elif IsEmpty(bottopprod[i]) then
	  fact := [];
	  for j in [1..Size(topbotprod[i])] do
	    Add(fact,[-topbotprod[i][j]]);
	  od;
	  Append(midfactors,fact);
    elif Size(topbotprod[i]) = Size(bottopprod[i]) then
	  fact := [];
	  Add(fact,[-topbotprod[i][1],bottopprod[i][1]]);
	  for j in [2..Size(topbotprod[i])] do
		Add(fact,[-topbotprod[i][j]]);
		Add(fact,[bottopprod[i][j]]);
	  od;
	  Append(midfactors,fact);
	elif Size(topbotprod[i]) < Size(bottopprod[i]) then
	  fact := [];
	  Add(fact,[-topbotprod[i][1],bottopprod[i][1]]);
	  for j in [2..Size(topbotprod[i])] do
		Add(fact,[-topbotprod[i][j]]);
		Add(fact,[bottopprod[i][j]]);
	  od;
	  for j in [Size(topbotprod[i])+1..Size(bottopprod[i])] do 
		Add(fact,[bottopprod[i][j]]);
	  od;
	  Append(midfactors,fact);
	elif Size(topbotprod[i]) > Size(bottopprod[i]) then
	  fact := [];
	  Add(fact,[-topbotprod[i][1],bottopprod[i][1]]);
	  for j in [2..Size(bottopprod[i])] do
		Add(fact,[-topbotprod[i][j]]);
		Add(fact,[bottopprod[i][j]]);
	  od;
	  for j in [Size(bottopprod[i])+1..Size(topbotprod[i])] do
		Add(fact,[-topbotprod[i][j]]);
	  od;
	  Append(midfactors,fact);
	fi;
  od;
  count := count + pRoFactCount(Bipartition(midfactors));
  
  return count;
end;
  
LengthPaFact := function(n,runs)
  local bipartition,list,a;
  list := [];
  for a in [1..runs] do
    bipartition := PaRandom(n);
    Add(list,PaFactCount(bipartition));
  od;
  return Float(Average(list));
end;
